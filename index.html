async function handleChat() { const inputField = document.getElementById('userInput'); const chatFlow = document.getElementById('chat-flow'); const sendBtn = document.getElementById('sendBtn'); const text = inputField.value.trim(); if (text === "") return; addBubble(text, 'user-bubble'); inputField.value = ""; sendBtn.disabled = true; const loadingBubble = addBubble("در حال دریافت پاسخ از ماورای ذهن...", 'ai-bubble'); try { // تغییر آدرس از v1beta به v1 برای پایداری بیشتر const response = await fetch(`https://generativelanguage.googleapis.com/v1/models/gemini-1.5-flash:generateContent?key=${API_KEY}`, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ contents: [{ parts: [{ text: "تو دستیار هوشمند سایت PlanLife هستی. لحن تو مثبت، انگیزشی و کوتاه است. پاسخ بده به: " + text }] }] }) }); const data = await response.json(); if (data.error) { loadingBubble.innerText = "خطای فنی: " + data.error.message; } else if (data.candidates && data.candidates[0].content) { loadingBubble.innerText = data.candidates[0].content.parts[0].text; } else { loadingBubble.innerText = "پاسخی دریافت نشد. دوباره امتحان کن."; } } catch (error) { loadingBubble.innerText = "ارتباط با سرور قطع شد. لطفاً وضعیت VPN را چک کنید."; console.error(error); } finally { sendBtn.disabled = false; chatFlow.scrollTop = chatFlow.scrollHeight; } }
